\subsection{The Complete Training Iteration}
\label{sec:iteration}

\todo{Complete lead-in sentence ending in a colon, e.g.:}
\Cref{alg:fsdp} summarises one FSDP training iteration on rank $i$:

\begin{algorithm}[ht]
  \caption{One FSDP training iteration on rank $i$}
  \label{alg:fsdp}
  \begin{algorithmic}[1]
    \Require Parameter shards $W^{(u)}_i$ and optimizer-state shards for
      every unit $u = 1,\dots,U$; local mini-batch $x^{(i)}$
    \Statex \textit{// Forward pass}
    \For{$u = 1$ \textbf{to} $U$}
      \State $W^{(u)} \gets \textsc{AllGather}\big(W^{(u)}_i\big)$
      \State Compute forward of unit $u$ with $W^{(u)}$
      \State Free all of $W^{(u)}$ except $W^{(u)}_i$ \Comment{reshard}
    \EndFor
    \State Compute loss $\mathcal{L}$
    \Statex \textit{// Backward pass}
    \For{$u = U$ \textbf{down to} $1$}
      \State $W^{(u)} \gets \textsc{AllGather}\big(W^{(u)}_i\big)$
      \State Compute local gradient $g^{(i),u}$ of unit $u$
      \State Free all of $W^{(u)}$ except $W^{(u)}_i$
      \State $g^{(u)}_i \gets \textsc{ReduceScatter}\big(g^{(i),u}\big) / N$
      \State Free $g^{(i),u}$
    \EndFor
    \Statex \textit{// Optimizer step (local, no communication)}
    \For{$u = 1$ \textbf{to} $U$}
      \State Update $W^{(u)}_i$ using $g^{(u)}_i$ and its optimizer-state shard
    \EndFor
  \end{algorithmic}
\end{algorithm}

  \label{eq:mem-ddp}

\begin{proposition}
\label{prop:mem-fsdp}
Under full sharding across $N$ GPUs, the persistent model-state memory
per GPU is
\begin{equation}
  M_{\text{FSDP}} = \frac{16\Psi}{N} \text{ bytes.}
  \label{eq:mem-fsdp}
\end{equation}
\end{proposition}

  \caption{FSDP sharding strategies. \todo{check wording}}
  \label{tab:strategies}
  \small
  \begin{tabular}{@{}l>{\raggedright}p{3.2cm}c>{\raggedright\arraybackslash}p{5.2cm}@{}}
    \toprule
    Strategy & Sharded state & Reshard after fwd? & Memory / communication effect \\
    \midrule
    \texttt{FULL\_SHARD}    & Params, grads, optimizer state & Yes
      & Lowest memory; $\sim 3\Psi$ communication \\
    \texttt{SHARD\_GRAD\_OP} & Grads, optimizer state & No

      & Equivalent to DDP; $16\Psi$ per GPU, $\sim 2\Psi$ communication \\

      & Memory reduced by the intra-node group size; slow inter-node links only carry gradient All-Reduce \\
    \bottomrule
  \end{tabular}
\end{table}
