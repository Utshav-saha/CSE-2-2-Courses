\section{Pseudocode}
\begin{algorithm}[ht]
\caption{Calculate Factorial}
\label{alg:factorial}
\begin{algorithmic}[1] % The [1] adds line numbers to every line
    \Require $n \geq 0$
    \Ensure Returns $n!$

    \State $result \gets 1$

    \If{$n = 0$ \textbf{or} $n = 1$}
        \State \Return $result$
    \EndIf

    \While{$n > 1$}
        \State $result \gets result \times n$
        \State $n \gets n - 1$
    \EndWhile

    \State \Return $result$
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[ht]
\caption{Ray--Scene Intersection (new example)}
\label{alg:raycast}
\begin{algorithmic}[1]
    \Require ray $r$, scene $S$
    \Ensure nearest hit $h$ or $\varnothing$
    \State $t_{\min} \gets \infty$; \quad $h \gets \varnothing$
    \ForAll{objects $o \in S$}
        \State $t \gets \Call{Intersect}{r, o}$
        \If{$t > 0$ \textbf{and} $t < t_{\min}$}
            \State $t_{\min} \gets t$
            \State $h \gets o$
        \EndIf
    \EndFor
    \State \Return $h$
\end{algorithmic}
\end{algorithm}

\section{Introduction to Image Synthesis}
Image synthesis is the process of generating images from mathematical and algo-rithmic descriptions. Modern graphics systems rely on \textbf{precise mathematical models} and efficient computation. \emph{Even small numerical errors} can result in visible artifacts in rendered images. Typical formulations involve scalar values such as $x$, indexed parameters
like $I_{ij}$, and exponential terms $e^{x^2}$. These elements are combined to simulate realistic lighting behavior.

\subsection{Theorems and definitions (new)}

\begin{definition}[Radiance]
\label{def:radiance}
Radiance $L$ is the radiant flux per unit projected area per unit solid angle,
measured in \si{\watt\per\square\metre\per\steradian}.
\end{definition}

\begin{theorem}[Energy conservation]
\label{thm:energy}
For any physically valid BRDF $f_r$ and all outgoing directions $\bm{\omega}_o$,
\[
\int_{\Omega} f_r(\bm{x},\bm{\omega}_i,\bm{\omega}_o)\,
(\bm{\omega}_i \cdot \bm{n}) \, \mathrm{d}\bm{\omega}_i \le 1 .
\]
\end{theorem}

\begin{proof}
The reflected flux cannot exceed the incident flux; dividing both sides by the
incident flux and applying Definition~\ref{def:radiance} gives the bound. Equality
holds only for a lossless surface.
\end{proof}

\begin{lemma}
A Lambertian BRDF $f_r = \rho/\pi$ satisfies Theorem~\ref{thm:energy} iff
$\rho \le 1$.
\end{lemma}

\begin{remark}
In practice $\rho \in [0.02, 0.9]$ for real-world dielectrics.
\end{remark}
