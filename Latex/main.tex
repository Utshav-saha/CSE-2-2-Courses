\item We want to write the solution for the quadratic equation $$\alpha x^2 + bx + c = 0$$. 

The solution is 
$$ x = \frac{1}{2\alpha} \times \left(-b + \sqrt{-b^2 \pm 4\cdot \alpha \cdot c}\right) $$

But there are two solutions actually.
\begin{align*}
 x_1 & = \frac{1}{2\alpha} \times \left(-b + \sqrt{-b^2 + 4\cdot \alpha \cdot c}\right)  \\ 
 x_2 & = \frac{1}{2\alpha} \times \left(-b - \sqrt{-b^2 - 4\cdot \alpha \cdot c}\right) \\
 & = \frac{-b - \sqrt{-b^2 - 4 \alpha c}}{2a}
\end{align*}
\end{enumerate}

\section{Pseudocode}
\begin{algorithm}[ht]
\caption{Calculate Factorial}
\label{alg:factorial}
\begin{algorithmic}[1] % The [1] adds line numbers to every line
    \Require $n \geq 0$
    \Ensure Returns $n!$
    
    \State $result \gets 1$
    
    \If{$n = 0$ \textbf{or} $n = 1$}
        \State \Return $result$
    \EndIf
    
    \While{$n > 1$}
        \State $result \gets result \times n$
        \State $n \gets n - 1$
    \EndWhile
    
    \State \Return $result$
\end{algorithmic}
\end{algorithm}

def calculate_fibonacci(n):
    """Calculate the nth Fibonacci number."""
    if n <= 0:
        return 0
    elif n == 1:
        return 1
    else:
        return calculate_fibonacci(n-1) + calculate_fibonacci(n-2)

print(calculate_fibonacci(10))
\end{lstlisting}
